cmake_minimum_required(VERSION 3.24)
project(linscaler VERSION 0.1.0 DESCRIPTION "Coordinate and index mathematics for uniform grids" LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(linscaler_core INTERFACE)
add_library(linscaler::core ALIAS linscaler_core)
set_target_properties(linscaler_core PROPERTIES EXPORT_NAME core)
target_compile_features(linscaler_core INTERFACE cxx_std_11)
target_include_directories(linscaler_core INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
add_library(linscaler_transform INTERFACE)
add_library(linscaler::transform ALIAS linscaler_transform)
set_target_properties(linscaler_transform PROPERTIES EXPORT_NAME transform)
target_link_libraries(linscaler_transform INTERFACE linscaler::core)
target_compile_features(linscaler_transform INTERFACE cxx_std_11)
add_library(linscaler INTERFACE)
add_library(linscaler::linscaler ALIAS linscaler)
target_link_libraries(linscaler INTERFACE linscaler::transform)
target_compile_features(linscaler INTERFACE cxx_std_17)
option(LINSCALER_BUILD_TESTS "Build the test suite" ${PROJECT_IS_TOP_LEVEL})
option(LINSCALER_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(LINSCALER_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(LINSCALER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

# Development flags belong to our executables, never to consumers of the library.
function(linscaler_development_target target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(LINSCALER_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(LINSCALER_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(LINSCALER_SANITIZERS)
        if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            message(FATAL_ERROR "Sanitizer preset requires GCC/Clang on a Unix host")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

if(LINSCALER_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(LINSCALER_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
install(DIRECTORY include/linscaler DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/linscaler)
install(TARGETS linscaler linscaler_core linscaler_transform EXPORT linscaler-targets)
install(EXPORT linscaler-targets FILE linscaler-targets.cmake NAMESPACE linscaler::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/linscaler)
configure_package_config_file(cmake/linscaler-config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/linscaler-config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/linscaler)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/linscaler-config-version.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion ARCH_INDEPENDENT)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/linscaler-config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/linscaler-config-version.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/linscaler)
