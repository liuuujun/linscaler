cmake_minimum_required(VERSION 3.24)
project(linscaler_consumer LANGUAGES CXX)
# The full target must raise this to C++17; core and transform must leave it at 11.
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CONSUMER_MODE STREQUAL "subdirectory")
    add_subdirectory("${LINSCALER_SOURCE}" linscaler)
    if(TARGET test_legacy OR TARGET example_grid_mapping)
        message(FATAL_ERROR "Subproject enabled development executables by default")
    endif()
else()
    find_package(linscaler 0.1 CONFIG REQUIRED)
endif()
foreach(library linscaler::core linscaler::transform linscaler::linscaler)
    foreach(property INTERFACE_COMPILE_OPTIONS INTERFACE_LINK_OPTIONS)
        get_target_property(options ${library} ${property})
        if(options)
            message(FATAL_ERROR "${library} leaked ${property}: ${options}")
        endif()
    endforeach()
endforeach()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE linscaler::linscaler)
add_executable(consumer_core main_core.cpp)
target_link_libraries(consumer_core PRIVATE linscaler::core)
add_executable(consumer_transform main_transform.cpp)
target_link_libraries(consumer_transform PRIVATE linscaler::transform)
foreach(consumer consumer_core consumer_transform)
    if(MSVC)
        # No strict C++11 mode is available on MSVC.
        set_target_properties(${consumer} PROPERTIES CXX_STANDARD 14)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${consumer} PRIVATE -pedantic-errors)
    endif()
endforeach()
enable_testing()
add_test(NAME consume COMMAND consumer)
add_test(NAME consume_core COMMAND consumer_core)
add_test(NAME consume_transform COMMAND consumer_transform)
